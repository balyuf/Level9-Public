.SetCGAPalettes
 cif pc
; dc.b $0,$F,$F,$F,$F,$F,$0,$0,$F,$0,$9,$F,$F,$F,$0,$F ; 2 colour text printing
; dc.b $0,$F,$F,$F,$F,$F,$0,$0,$F,$0,$9,$F,$F,$F,$0,$F ; 2 color map

  v1=4 ; list 4
  &v2=list4(20) ; CGA2 palette
  v3=2 ; 2-color palette
  v4=255 ; 255 is text printing palette
  gosub @MCSetPalette

  add v2,c16 ; skip on to default palette (map)
  v4=0 ; default palette
  gosub @MCSetPalette

  add v2,c16 ; skip on to people palette
  v4=1 ; palette 1 - people
  gosub @MCSetPalette


  v1=4 ; res-set up list just in case
  &v2=list4(22) ; CGA4 palette
  v3=4 ; 4-color palette
  v4=255 ; 255 is text printing palette
  gosub @MCSetPalette

  add v2,c16 ; skip on to default palette (map)
  v4=0 ; default palette
  gosub @MCSetPalette

  add v2,c16 ; skip on to people palette
  v4=1 ; palette 1 - people
  gosub @MCSetPalette
cend

 cend
